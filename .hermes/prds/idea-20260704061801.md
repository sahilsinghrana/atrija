## 1. Overview

**One-liner:** Automated accessibility testing using axe-core to run on every build and report violations.

**Problem:** Accessibility issues can creep in unnoticed during development, leading to a poor experience for users relying on assistive technologies. Manual accessibility testing is infrequent and prone to oversight.

**Solution:** Integrate axe-core into the project's testing suite (using vitest or a dedicated script) to automatically scan the rendered DOM for accessibility violations according to WCAG 2.1 standards. Fail the build if violations above a certain threshold are detected.

## 2. User Stories

- As a developer, I want accessibility tests to run automatically so I can catch issues early.
- As a user with disabilities, I want the site to remain accessible so I can navigate and enjoy the content without barriers.
- As the product owner, I want to maintain a high accessibility score to comply with standards and broaden our audience.

## 3. Technical Specification

### 3.1 Architecture

- **New script:** `scripts/test-accessibility.js` – Uses axe-core to run accessibility checks on a built version of the site.
- **Integration:** Add a new npm script `test:accessibility` that runs the script after building the site in a temporary directory.
- **Configuration:** Allow rule tagging (e.g., exclude certain rules if false positives) and set a violation threshold (e.g., fail on any violation of severity 'serious' or higher).
- **Reporting:** Output a detailed report to the console and optionally generate an HTML report for debugging.

#### Step 2: Create test-accessibility.js

```javascript
// scripts/test-accessibility.js
const { execSync } = require('child_process');
const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');
const { AxePuppeteer } = require('@axe-core/puppeteer');
const puppeteer = require('puppeteer');

const REPORT_DIR = path.join(__dirname, '..', 'test', 'accessibility-reports');
const THRESHOLD_SEVERITY = 'moderate'; // fail on moderate or worse

async function runAccessibilityTest(url) {
  const browser = await puppeteer.launch({
    args: ['--no-sandbox', '--disable-setuid-sandbox'],
    headless: true,
  });
  const page = await browser.newPage();
  await page.goto(url, { waitUntil: 'networkidle0' });

  const axeSource = await fs.promises.require.resolve('axe-core/axe.min.js');
  await page.addScriptTag({ path: axeSource });

  const results = await page.evaluate(async () => {
    return await axe.run({
      // Optionally configure rules, tags, etc.
      // runOnly: {
      //   type: 'tag',
      //   values: ['wcag2a', 'wcag21aa']
      // }
    });
  });

  await browser.close();
  return results;
}

function formatViolations(results) {
  const { violations } = results;
  if (violations.length === 0) return 'No accessibility violations found.';

  let output = `Found ${violations.length} accessibility violation(s):\n\n`;
  violations.forEach((v) => {
    output += `• ${v.description}\n`;
    output += `  Help: ${v.help}\n`;
    output += `  Nodes: ${v.nodes.length}\n`;
    v.nodes.forEach((node) => {
      output += `    - ${node.target.join(', ')}: ${node.all[0].message}\n`;
    });
    output += '\n';
  });
  return output;
}

async function main() {
  try {
    // Build the site first
    console.log('Building site for accessibility testing...');
    execSync('npm run build', { stdio: 'inherit' });

    // Serve the built site on a random port
    const port = 4000 + Math.floor(Math.random() * 1000);
    const server = require('http').createServer((req, res) => {
      const fs = require('fs');
      const path = require('path');
      const filePath = path.join(__dirname, '..', 'dist', req.url === '/' ? 'index.html' : req.url);
      fs.stat(filePath, (err, stats) => {
        if (err || !stats.isFile()) {
          res.statusCode = 404;
          res.end('Not Found');
          return;
        }
        fs.createReadStream(filePath).pipe(res);
      });
    });

    server.listen(port, () => {
      console.log(`Serving at http://localhost:${port}`);
    });

    // Wait a moment for server to start
    await new Promise(resolve => setTimeout(resolve, 1000));

    const results = await runAccessibilityTest(`http://localhost:${port}`);
    server.close();

    const output = formatViolations(results);
    console.log(output);

    // Determine if we should fail based on severity
    const hasFailures = results.violations.some(v => {
      const severityLevels = ['minor', 'moderate', 'serious', 'critical'];
      return severityLevels.indexOf(v.impact) >= severityLevels.indexOf(THRESHOLD_SEVERITY);
    });

    process.exit(hasFailures ? 1 : 0);
  } catch (err) {
    console.error('Accessibility test failed:', err);
    process.exit(1);
  }
}

module.exports = { main };
if (require.main === module) {
  main();
}
```

## 6. Dependencies and Risks

**Dependencies:** axe-core, puppeteer, nodejs.

**Risks:**
- Puppeteer requires a compatible Chromium binary; ensure it's available in the CI environment.
- Accessibility tests can be slow; consider running them only on PRs or nightly builds if they significantly increase CI time.
- False positives may occur; allow configuring rule exclusions.
